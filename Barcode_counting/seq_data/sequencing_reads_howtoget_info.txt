if (( $# > 0 )); then
    RUNS=("$@")

THREADS=8
SRA_DIR="$PWD/sra_cache"
FASTQ_DIR="$PWD/downloaded_fastq"
TMP_DIR="$PWD/sra_tmp"

die() { printf '%s\n' "$*" >&2; exit 1; }

(( ${#RUNS[@]} > 0 )) || die 'No SRA runs selected.'
for run in "${RUNS[@]}"; do
    [[ "$run" =~ ^SRR[0-9]+$ ]] || die "Invalid SRA run accession: $run"
done
for tool in prefetch fasterq-dump gzip find; do
    command -v "$tool" >/dev/null || die "Required tool missing: $tool"
done
mkdir -p "$SRA_DIR" "$FASTQ_DIR" "$TMP_DIR"

# Look for the archive file itself, including the SRA Lite format.
find_archive() {
    local accession="$1" suffix found
    for suffix in sra sralite; do
        found=$(find "$SRA_DIR" -type f -name "${accession}.${suffix}" -print -quit)
        if [[ -n "$found" ]]; then
            printf '%s\n' "$found"
            return 0
        fi
    done
    return 1
}

for run in "${RUNS[@]}"; do
    # Some toolkit versions retain the archive extension in FASTQ names.
    complete=false
    for stem in "$run" "$run.sra" "$run.sralite"; do
        if [[ -s "$FASTQ_DIR/${stem}_1.fastq.gz" && -s "$FASTQ_DIR/${stem}_2.fastq.gz" ]]; then
            complete=true
            break
        fi
    done
    if "$complete"; then
        printf 'Already converted: %s\n' "$run"
        continue
    fi

    # Preserve incomplete outputs for inspection rather than overwriting them.
    for stem in "$run" "$run.sra" "$run.sralite"; do
        for suffix in _1.fastq _2.fastq _1.fastq.gz _2.fastq.gz; do
            [[ ! -e "$FASTQ_DIR/${stem}${suffix}" ]] ||
                die "Existing incomplete FASTQ output for $run: $FASTQ_DIR/${stem}${suffix}. Inspect it before retrying."
        done
    done

    if ! archive=$(find_archive "$run"); then
        prefetch "$run" --max-size u -O "$SRA_DIR/$run"
        archive=$(find_archive "$run") || die "No .sra or .sralite archive found for $run."
    else
        printf 'Using cached archive: %s\n' "$archive"
    fi

    fasterq-dump "$archive" --split-files -e "$THREADS" \
        -t "$TMP_DIR" -O "$FASTQ_DIR"

    for fastq in "$FASTQ_DIR/${run}"*.fastq; do
        if [[ -f "$fastq" ]]; then
            gzip -- "$fastq"
        fi
    done
done
